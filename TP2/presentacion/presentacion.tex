\section{Introduccion}

\begin{frame}
  \centering
  \Large Introducci\'on / Sistema Real / Fundamentos
\end{frame}

\begin{frame}{Sistema real}
  Las \textbf{bandadas de agentes autopropulsados} (bancos de peces, bandadas
  de aves, enjambres de bacterias) son sistemas de muchas part\'iculas que se
  mueven a rapidez aproximadamente constante y ajustan continuamente su
  direcci\'on seg\'un las de sus vecinos cercanos.

  \vspace{0.5cm}

  Sin l\'ider ni reglas globales, emerge un comportamiento colectivo
  \textbf{ordenado} (movimiento coherente) o \textbf{desordenado} seg\'un el
  nivel de ruido en la interacci\'on -- una transici\'on de fase
  orden-desorden.
\end{frame}

\begin{frame}{Modelo matem\'atico general}
  Cada part\'icula $i$ tiene posici\'on $\vect{r}_i(t)$ y orientaci\'on
  $\theta_i(t)$, movi\'endose a rapidez constante $v_0$:

  \[
    \vect{r}_i(t+\Delta t) = \vect{r}_i(t) + v_0\,(\cos\theta_i(t+\Delta t),\,
    \sin\theta_i(t+\Delta t))\,\Delta t
  \]

  La orientaci\'on se actualiza seg\'un la interacci\'on con el vecindario
  $\mathcal{N}_i$ (part\'iculas dentro de un radio $r_c$) m\'as un ruido
  angular $\eta$:

  \begin{itemize}
    \item \textbf{Vicsek}: promedio circular de las orientaciones de
      $\mathcal{N}_i$
      \[
        \theta_i(t+\Delta t) = \operatorname{atan2}\!\Big(\textstyle\sum_{j\in\mathcal{N}_i}\sin\theta_j,\;
        \sum_{j\in\mathcal{N}_i}\cos\theta_j\Big) + \Delta\theta_i
      \]
    \item \textbf{Votante}: copia la orientaci\'on de un \'unico vecino
      elegido al azar
      \[
        \theta_i(t+\Delta t) = \theta_j(t) + \Delta\theta_i,\qquad
        j \sim \mathcal{U}(\mathcal{N}_i)
      \]
  \end{itemize}

  En ambos casos $\Delta\theta_i \sim \mathcal{U}(-\eta/2,\ \eta/2)$.
\end{frame}

\section{Implementacion}

\begin{frame}

\begin{frame}{Arquitectura del motor y esquema del \texttt{step()}}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}
        \item \texttt{VicsekParticle}: posici\'on $(x,y)$ + orientaci\'on
          $\theta$ (extiende la part\'icula de TP1, sin velocidad).
        \item \texttt{Grid}: CIM de TP1 reutilizado, ahora
          \textbf{persistente} (buffers reusados en \texttt{rebuild()}).
        \item \texttt{Simulation}: encapsula part\'iculas, grilla, RNG.
        \item Interacci\'on seleccionable: \texttt{enum class Model
          \{ Vicsek, Voter \}}.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \texttt{step()}, sincr\'onico (double-buffered):
      \begin{enumerate}
        \item \texttt{grid\_.rebuild()} sobre la foto previa al paso.
        \item Nueva orientaci\'on de \emph{todas} las part\'iculas
          (Vicsek/votante + ruido) a un buffer auxiliar.
        \item Avanzar posici\'on con la nueva orientaci\'on, PBC, y
          reci\'en ah\'i confirmar \texttt{theta}.
      \end{enumerate}
      $v_a$ y $S$ se calculan sobre la misma lista de adyacencia del CIM.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Geometr\'ia y parametros}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Caja cuadrada $L=10$, condiciones peri\'odicas de contorno.
        \item Densidades $\rho \in \{2,4,8\}$.
        \item Ambos modelos: Vicsek y votante.
        \item Variable de estudio: ruido angular $\eta$.
        \item Grilla de $\eta$: gruesa (9 pts en $[0,2\pi]$) + fina (8 pts)
          localizada por un mini-barrido exploratorio del bracket de
          transici\'on, por cada (modelo, $\rho$).
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \includegraphics[width=\textwidth]{frame_vicsek_rho2.png}

      \small $\rho=2$, esquema de la caja $L\times L$ con vectores de
      velocidad por part\'icula
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Observables y repeticiones}
  Polarizaci\'on (parametro de orden global):
  \[
    v_a = \left| \frac{1}{N} \sum_{i=1}^{N} \hat{\vect{v}}_i \right|
  \]

  Fracci\'on del cluster gigante ($S$): tama\~no de la componente conexa
  m\'as grande de la red de vecinos (CIM, radio $r_c$) sobre $N$:
  \[
    S = \frac{|\text{componente conexa m\'as grande}|}{N}
  \]

  \vspace{0.3cm}
  \textbf{Repeticiones:} $K \geq 5$ semillas independientes por punto
  $(\text{modelo}, \rho, \eta)$, semilla determin\'istica
  (\texttt{sha256}), nunca por reloj. \\
  \textbf{Duraci\'on:} 2000 pasos por corrida; estado estacionario = \'ultimo
  50\% de los pasos (mismo criterio para $v_a$ y $S$). \\
  \textbf{Tiempo real medido} (Plan de benchmark, $N=1000$):
  $\approx 1.58$~ms por paso completo del motor de TP2 (rebuild de grilla +
  b\'usqueda de vecinos + integraci\'on + escritura de trayectoria).
\end{frame}

\section{Conclusiones}

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item Ambos modelos muestran transici\'on orden-desorden en $\eta$,
      pero Vicsek sostiene el orden hasta $\eta$ mucho mayores que el
      votante ($\eta_c$ de $2.58$ a $4.71$ vs. $\leq 0.22$).
    \item $\eta_c$ de Vicsek crece con la densidad; el votante permanece
      cerca de $\eta=0$ en las tres densidades.
    \item $S$ se mantiene cerca de 1 en casi todo el rango, salvo Vicsek
      $\rho=2$ (baja densidad), donde se acopla con la p\'erdida de $v_a$.
    \item El motor de TP2 preserva el escalado del CIM de TP1: el paso
      completo (incl. escritura, a mayor densidad $L=10$ vs $L=20$) es
      m\'as costoso en t\'erminos absolutos, pero ambas series crecen con
      $N$ con pendiente similar en log-log.
  \end{itemize}
\end{frame}
