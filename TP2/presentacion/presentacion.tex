\begin{frame}{Sistema real}
  Las \textbf{bandadas de agentes autopropulsados} (bancos de peces, bandadas
  de aves, enjambres de bacterias) son sistemas de muchas part\'iculas que se
  mueven a rapidez aproximadamente constante y ajustan continuamente su
  direcci\'on seg\'un las de sus vecinos cercanos.

  \vspace{0.5cm}

  Sin l\'ider ni reglas globales, emerge un comportamiento colectivo
  \textbf{ordenado} (movimiento coherente) o \textbf{desordenado} seg\'un el
  nivel de ruido en la interacci\'on -- una transici\'on de fase
  orden-desorden.
\end{frame}

\begin{frame}{Modelo matem\'atico general}
  Cada part\'icula $i$ tiene posici\'on $\vect{r}_i(t)$ y orientaci\'on
  $\theta_i(t)$, movi\'endose a rapidez constante $v_0$:

  \[
    \vect{r}_i(t+\Delta t) = \vect{r}_i(t) + v_0\,(\cos\theta_i(t+\Delta t),\,
    \sin\theta_i(t+\Delta t))\,\Delta t
  \]

  La orientaci\'on se actualiza seg\'un la interacci\'on con el vecindario
  $\mathcal{N}_i$ (part\'iculas dentro de un radio $r_c$) m\'as un ruido
  angular $\eta$:

  \begin{itemize}
    \item \textbf{Vicsek}: promedio circular de las orientaciones de
      $\mathcal{N}_i$
      \[
        \theta_i(t+\Delta t) = \operatorname{atan2}\!\Big(\textstyle\sum_{j\in\mathcal{N}_i}\sin\theta_j,\;
        \sum_{j\in\mathcal{N}_i}\cos\theta_j\Big) + \Delta\theta_i
      \]
    \item \textbf{Votante}: copia la orientaci\'on de un \'unico vecino
      elegido al azar
      \[
        \theta_i(t+\Delta t) = \theta_j(t) + \Delta\theta_i,\qquad
        j \sim \mathcal{U}(\mathcal{N}_i)
      \]
  \end{itemize}

  En ambos casos $\Delta\theta_i \sim \mathcal{U}(-\eta/2,\ \eta/2)$.
\end{frame}
