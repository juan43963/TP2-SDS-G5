\begin{frame}{Modelo matemático: tiempo al próximo choque}
  \textbf{Vuelo libre} --- entre choques:
  \[
    \vect{r}_i(t + t_c) = \vect{r}_i(t) + \vect{v}_i\, t_c
  \]

  \textbf{Pared vertical} ($x_{p1} < x_{p2}$):
  \[
    t_c = \frac{x_{p2} - r - x(0)}{v_x} \;\; (v_x > 0),
    \qquad
    t_c = \frac{x_{p1} + r - x(0)}{v_x} \;\; (v_x < 0)
  \]

  \textbf{Dos discos} $i$, $j$:
  \[
    t_c =
    \begin{cases}
      -\dfrac{\Delta\vect{v}\cdot\Delta\vect{r} + \sqrt{d}}{\Delta\vect{v}\cdot\Delta\vect{v}}
        & \Delta\vect{v}\cdot\Delta\vect{r} < 0,\; d \geq 0\\[6pt]
      \infty & \text{en otro caso}
    \end{cases}
    \qquad
    \begin{aligned}
      d &= (\Delta\vect{v}\cdot\Delta\vect{r})^2 - (\Delta\vect{v}\cdot\Delta\vect{v})(\Delta\vect{r}\cdot\Delta\vect{r} - \sigma^2)\\
      \sigma &= r_i + r_j
    \end{aligned}
  \]

  \vspace{0.2em}
  \centering
  El próximo evento es el \textbf{menor} $t_c$ entre todos los pares y paredes.
\end{frame}

% ---------------------------------------------------------------- 5
\begin{frame}{Modelo matemático: choques elásticos}
  \textbf{Pared} --- vertical: $(v_x, v_y) \to (-v_x, v_y)$, \quad horizontal: $(v_x, v_y) \to (v_x, -v_y)$

  \vspace{0.6em}
  \textbf{Dos discos} --- impulso a lo largo de la línea de centros:
  \[
    J = \frac{2\, m_i m_j\, (\Delta\vect{v}\cdot\Delta\vect{r})}{\sigma\,(m_i + m_j)},
    \qquad
    \vect{J} = \frac{J}{\sigma}\,(\Delta x,\, \Delta y)
  \]
  \[
    \vect{v}_i^{\,d} = \vect{v}_i^{\,a} + \frac{\vect{J}}{m_i},
    \qquad
    \vect{v}_j^{\,d} = \vect{v}_j^{\,a} - \frac{\vect{J}}{m_j}
  \]

  \textbf{Obstáculo fijo} --- límite $m_j \to \infty$, reflexión especular:
  \[
    \vect{v}^{\,d} = \vect{v}^{\,a} - 2\,(\vect{v}^{\,a}\cdot\hat{\vect{n}})\,\hat{\vect{n}}
  \]

  \vspace{0.2em}
  \centering
  {\footnotesize $a$: antes del choque, \ $d$: después, \ $\hat{\vect{n}}$: normal en el punto de contacto.}
\end{frame}

\begin{frame}{Observables}
  \textbf{Fracción de usadas} y \textbf{tiempo de llegada al 90\%} --- post-proceso del output:
  \[
    F_u(t) = \frac{N_g(t)}{N},
    \quad
    F_u(t_{90}) = 0{,}9,
    \quad
    \langle t_{90}\rangle = \frac{1}{n}\sum_{j=1}^{n} t_{90}^{(j)},
    \quad
    \sigma_{t_{90}} = \sqrt{\frac{1}{n-1}\sum_{j=1}^{n}\big(t_{90}^{(j)} - \langle t_{90}\rangle\big)^2}
  \]

  \textbf{Desplazamiento cuadrático medio} --- sobre las $N$ partículas, frescas y usadas:
  \[
    \mathrm{DCM}(t) = \frac{1}{N}\sum_{i=1}^{N} \big|\vect{r}_i(t) - \vect{r}_i(0)\big|^2
    \;=\; 4\,D\,t \quad \text{(régimen difusivo)}
  \]

  \vspace{0.3em}
  \hrule
  \vspace{0.5em}

  \params{
  Objetivo: minimizar $\langle t_{90}\rangle$. Barras: $\sigma_{t_{90}}$. Si una realización no llega a $F_u = 0{,}9$ antes de $t_{\max}$, no se promedia.\\[3pt]
  Realizaciones: 10 (tiempo de ejecución), 20 (exploración sistemática), 5 (búsqueda automática), 20 (comparación final, mismas semillas), 40 (validación de la elegida), 1 (DCM).\\[3pt]
  Tiempo simulado: $t_f = 30$ s (tiempo de ejecución), \ $t_{\max} = 100$ s (resto).
  }
\end{frame}

\begin{frame}{Hipótesis: área libre y acceso al arco}
  \textbf{Tasa de goles por partícula} (gas diluido en 2D):
  \[
    \nu \approx \frac{v_0 \; 2d}{\pi\, A_{\mathrm{libre}}}
  \]
  \begin{itemize}
    \item $v_0$ y $d$ son fijos: sólo los obstáculos cambian $A_{\mathrm{libre}}$.
    \item Menos área libre $\Rightarrow$ más visitas al arco, siempre que toda región accesible tenga un arco.
    \item Si la densidad crece demasiado dominan los choques entre partículas.
    \vspace{0.4em}
    \item \textbf{Prueba:} una pared que divide la mesa en dos cámaras, cada una con su arco, y se va engrosando.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- 29
\begin{frame}{Conclusiones}
  \begin{itemize}
    \item El tiempo de ejecución crece con $N$ más rápido que la cantidad de eventos procesados.
    \vspace{0.3em}
    \item Obstáculos aislados, embudos, compuertas o un canal no bajan $\langle t_{90}\rangle$ por debajo de la mesa vacía más allá del desvío.
    \vspace{0.3em}
    \item Dividir la mesa en dos cámaras, cada una con su arco, sí lo baja, y cuanto más gruesa la pared, más.
    \vspace{0.3em}
    \item $\langle t_{90}\rangle$ tiene un mínimo para cámaras de $\approx 2{,}5\times10^{-1}$ m y vuelve a crecer para cámaras más cortas.
    \vspace{0.3em}
    \item La configuración elegida baja $\langle t_{90}\rangle$ de $(22 \pm 2)$ s a $(14 \pm 2)$ s.
    \vspace{0.3em}
    \item La configuración más rápida no tiene régimen difusivo identificable; entre las demás no se observa una correlación clara entre $D$ y $\langle t_{90}\rangle$.
  \end{itemize}
\end{frame}
