% SLIDE sim_observables | V:B | T:30
\begin{frame}{Observables}
  \small
  \begin{itemize}
    \item $N_{in}(t)$: filas del frame con $y \ge 0$; $t_{evac}$: instante del último cruce, leído
      de los frames.
    \item $f_{TSDMD}(t) = \#\{i \mid \mathrm{scheme}_i=0\}\,/\,N_{in}(t)$.
    \item Promedio y error: $\langle X\rangle=\frac1n\sum_m X_m$, \quad
      $\mathrm{SEM}=s/\sqrt n$, solo sobre corridas completas.
    \item Distancia contra el TSDMD:
      $z=\dfrac{\langle t\rangle-\langle t\rangle_{TSDMD}}{\sqrt{\mathrm{SEM}^2+\mathrm{SEM}_{TSDMD}^2}}$;
      $|z|<2$: no se distingue.
    \item Tiempo de ejecución $W$ y pendiente log-log de $W$ contra $N$: ajuste lineal de $\log W$ contra $\log N$ sobre las medias de los cinco $N$.
  \end{itemize}
\end{frame}
